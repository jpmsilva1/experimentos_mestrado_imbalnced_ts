% ── Section Formatting ────────────────────────────────────────────────────────
\usepackage{titlesec}

% Section: large, orange number, dark title (= was "chapter" / h1 from md)
\titleformat{\section}
  {\normalfont\LARGE\bfseries\color{ClaudeDark}}
  {\textcolor{ClaudeOrange}{\thesection}}{0.8em}{}
  [\vspace{2pt}\textcolor{ClaudeRule}{\titlerule[0.5pt]}]

\titleformat{\subsection}
  {\normalfont\large\bfseries\color{ClaudeDark}}
  {\textcolor{ClaudeOrange}{\thesubsection}}{0.8em}{}

\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\color{ClaudeMuted}}
  {\thesubsubsection}{0.8em}{}

\titleformat{\paragraph}[runin]
  {\normalfont\normalsize\bfseries\color{ClaudeDark}}
  {}{0em}{}[---\enspace]

\titlespacing*{\section}{0pt}{2ex plus 0.5ex}{1ex plus 0.2ex}
\titlespacing*{\subsection}{0pt}{1.5ex plus 0.3ex}{0.5ex}
\titlespacing*{\subsubsection}{0pt}{1ex}{0.3ex}
